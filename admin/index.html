async function autoTranslate(sourceId, targetId, fromLang, toLang) { const sourceEl = document.getElementById(sourceId); const targetEl = document.getElementById(targetId); if(!sourceEl || !targetEl) return; // Го земаме целиот текст заедно со HTML форматирањето let text = sourceEl.tagName === 'TEXTAREA' ? sourceEl.value : sourceEl.innerHTML; text = text.trim(); if(!text) { alert('Нема текст за преведување!'); return; } const btn = event.currentTarget; const origText = btn.innerText; btn.innerText = 'Преведувам...'; btn.disabled = true; try { // Користиме POST барање за да овозможиме превод на долги текстови (пр. цели блог постови) const url = `https://translate.googleapis.com/translate_a/single?client=gtx&sl=${fromLang}&tl=${toLang}&dt=t`; const res = await fetch(url, { method: 'POST', headers: { 'Content-Type': 'application/x-www-form-urlencoded', }, body: `q=${encodeURIComponent(text)}` }); const data = await res.json(); // Google враќа низа од сегменти, ги спојуваме во еден текст let translatedText = ''; if (data && data[0]) { data[0].forEach(item => { if (item[0]) translatedText += item[0]; }); } // Ги поправаме таговите доколку Google вметнал празни места (пр. < b > станува ) translatedText = translatedText.replace(/<\s*\/\s*/g, '/g, '>'); // Го внесуваме преводот во целното поле if(targetEl.tagName === 'TEXTAREA') { targetEl.value = translatedText; } else { targetEl.innerHTML = translatedText; } triggerFieldUpdates(); } catch(err) { alert('Грешка при комуникација со преведувачот.'); console.error(err); } finally { btn.innerText = origText; btn.disabled = false; } }
